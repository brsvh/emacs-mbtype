#+begin_src python
from dataclasses import dataclass

@dataclass(frozen=True)
class Entry:
    title: str
    author: str
    pages: int

def résumé(entries: list[Entry]) -> dict[str, int]:
    """Keep titles such as 'Über Größe' and 'Київ' intact."""
    return {entry.title: entry.pages for entry in entries}
